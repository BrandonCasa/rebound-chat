import React, {
  useState,
  useEffect,
  useRef,
  useCallback,
  useLayoutEffect
} from "https://esm.sh/react@18";
import TypesReact from "https://esm.sh/@types/react@18";
import ReactDOM from "https://esm.sh/react-dom@18";
import TypesReactDom from "https://esm.sh/@types/react-dom@18";
import { useInView } from "https://esm.sh/react-intersection-observer@9.0.0";

const INITIAL_COUNT = 20;
const CHUNK_SIZE = 10;

const InfiniteScrollUpList: React.FC = () => {
  const [isReady, setIsReady] = useState(false);
  const [items, setItems] = useState<number[]>(() =>
    Array.from({ length: INITIAL_COUNT }, (_, i) => i + 1)
  );

  const [isLoading, setIsLoading] = useState(false);

  const containerRef = useRef<HTMLDivElement | null>(null);
  const [rootEl, setRootEl] = useState<HTMLDivElement | null>(null);
  const prevScrollHeightRef = useRef(0);

  // Attach the container and keep it as the root for IntersectionObserver
  const setContainerRef = useCallback((node: HTMLDivElement | null) => {
    containerRef.current = node;
    if (node) {
      setRootEl(node);
    }
  }, []);

  const { ref: topSentinelRef, inView } = useInView({
    threshold: 0,
    root: rootEl ?? null
  });

  const loadMoreItems = useCallback(() => {
    if (isLoading) return;

    const container = containerRef.current;
    if (!container) return;

    setIsReady(false);
    setIsLoading(true);
    // Remember the old scrollHeight so we can adjust later
    prevScrollHeightRef.current = container.scrollHeight;

    setItems((prev) => {
      const currentTopValue = prev[prev.length - 1] ?? 0;
      const newItems = Array.from(
        { length: CHUNK_SIZE },
        (_, i) => currentTopValue + i + 1
      );
      return [...prev, ...newItems];
    });
  }, [isLoading]);

  // Trigger loading when the sentinel at the top comes into view
  useEffect(() => {
    if (inView && !isLoading && isReady) {
      loadMoreItems();
      setTimeout(() => {
        setIsReady(true);
      }, 200);
    }
  }, [inView, isLoading, loadMoreItems, isReady]);

  // After items change while loading, fix scroll position so it doesn't jump
  useLayoutEffect(() => {
    if (!isLoading) return;

    const container = containerRef.current;
    if (!container) return;

    const newScrollHeight = container.scrollHeight;
    const diff = prevScrollHeightRef.current - newScrollHeight - 20;

    // Because of column-reverse, adding items changes scrollHeight.
    // Adjust scrollTop so the user keeps their place.
    containerRef.current.scrollTo({
      top: containerRef.current.scrollTop - diff,
      behavior: "instant"
    });

    setIsLoading(false);
  }, [items, isLoading]);

  useEffect(() => {
    setIsReady(true);
  }, []);

  return (
    <div
      ref={setContainerRef}
      style={{
        height: "300px",
        border: "1px solid #ccc",
        overflowY: "auto",
        padding: "8px",
        boxSizing: "border-box",
        display: "flex",
        flexDirection: "column-reverse"
      }}
    >
      {items.map((item) => (
        <div
          key={item}
          style={{
            padding: "8px",
            marginTop: "4px",
            borderRadius: "4px",
            border: "1px solid #eee"
          }}
        >
          Basic item {item}
        </div>
      ))}

      {/* extra spacing at the very top */}
      <div style={{ height: "20px" }} />

      {/* sentinel that sits at the visual top because of column-reverse */}
      <div ref={topSentinelRef} />
    </div>
  );
};

ReactDOM.render(<InfiniteScrollUpList />, document.getElementById("root"));
